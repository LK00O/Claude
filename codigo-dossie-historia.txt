    function esc(s) { return String(s).replace(/&/g, "&amp;").replace(/"/g, "&quot;").replace(/</g, "&lt;").replace(/>/g, "&gt;"); }
    function pad(n) { return n < 10 ? "0" + n : "" + n; }

      clearTimeout(toastT); toastT = setTimeout(function () { el.classList.remove("show"); }, 2200);
    }

    function countTo(el, to) {
      var from = parseInt(el.getAttribute("data-v") || el.textContent, 10) || 0;
      el.setAttribute("data-v", to);
      if (reduceMotion || from === to) { el.textContent = to; return; }
      var start = null;
      var token = {}; el._t = token;
      (function step(ts) {
        if (el._t !== token) return;
        if (start === null) start = ts;
        var p = Math.min(1, (ts - start) / 450);
        el.textContent = Math.round(from + (to - from) * (1 - Math.pow(1 - p, 3)));
        if (p < 1) requestAnimationFrame(step);
      })(performance.now());
    }

    /* Imagens com cadeia de fallback */
    function imgTag(t, w, cls) {
      var list = sourcesFor(t, w);
      var rest = list.slice(1).map(function (s) { return s.src + "::" + s.cap; }).join("||");
      return '<img class="' + (cls || "") + '" src="' + esc(list[0].src) + '" alt="' + esc(t.caption) + '" loading="lazy" decoding="async" referrerpolicy="no-referrer" data-rest="' + esc(rest) + '" onerror="dossieImgFail(this)">';
    }
    window.dossieImgFail = function (img) {
      var rest = img.getAttribute("data-rest") || "";
      if (rest) {
        var parts = rest.split("||");
        var next = parts.shift().split("::");
        img.setAttribute("data-rest", parts.join("||"));
        img.src = next[0];
        if (next[1]) {
          img.alt = next[1];
          var fig = img.closest("figure");
          var cap = fig && fig.querySelector("figcaption");
          if (cap) cap.textContent = next[1];
        }
        return;
      }
      var media = img.closest(".card-media");
      if (media) {
        var fb = document.createElement("span");
        fb.className = "media-fallback";
        fb.textContent = img.alt || "Obra histórica";
        media.replaceChild(fb, img);
        return;
      }
      var btn = img.closest(".figure-btn");
      if (btn) {
        var plate = document.createElement("div");
        plate.className = "placard";
        plate.innerHTML = '<p>' + esc(img.alt || "Obra histórica") + '</p><a href="https://commons.wikimedia.org" target="_blank" rel="noopener noreferrer">Ver acervo da Wikimedia Commons ↗</a>';
        btn.parentNode.replaceChild(plate, btn);
        var fc = plate.parentNode.querySelector("figcaption");
        if (fc) fc.style.display = "none";
      }
    };

    var CIRC = 2 * Math.PI * 34;
    function updateProgress(animate) {
      var total = 0, done = 0;
      CHAPTERS.forEach(function (ch) {
        var d = doneCount(ch); total += ch.topics.length; done += d;
        var li = document.querySelector('[data-ch="' + ch.id + '"]');
        if (li) {
          li.querySelector(".ch-count").textContent = d + "/" + ch.topics.length;
          li.querySelector(".ch-bar span").style.transform = "scaleX(" + (d / ch.topics.length) + ")";
        }
      });
      var pct = Math.round(done / total * 100);
      $("ringBar").style.strokeDashoffset = (CIRC * (1 - done / total)).toFixed(2);
      if (animate) countTo($("ringPct"), pct); else { $("ringPct").textContent = pct; $("ringPct").setAttribute("data-v", pct); }
      $("ringMeta").textContent = done + " de " + total + " tópicos";
      $("ring").setAttribute("aria-label", "Progresso geral: " + pct + "%");

      var ch = chapter(state.ch), d2 = doneCount(ch);
      var cp = $("chapCount");
      if (cp) {
        cp.textContent = d2 + " de " + ch.topics.length + " concluídos";
        $("chapLine").style.transform = "scaleX(" + (d2 / ch.topics.length) + ")";
      }
    }

    function renderSidebar() {
      $("chapters").innerHTML = CHAPTERS.map(function (ch) {
        return '<li><button class="ch-btn" type="button" data-ch="' + ch.id + '" aria-current="' + (ch.id === state.ch) + '" style="--ch:' + ch.primary + '">' +
          '<span class="ch-num">Cap. ' + ch.num + '</span><span class="ch-count num"></span>' +
          '<span class="ch-name">' + ch.short + '</span>' +
          '<span class="ch-bar"><span></span></span></button></li>';
      }).join("");
      var fb = $("filters").querySelectorAll("button");
      for (var i = 0; i < fb.length; i++) fb[i].setAttribute("aria-pressed", fb[i].getAttribute("data-filter") === state.filter);
    }

    function renderBox(b) {
      if (b.type === "quote") return '<blockquote class="quote"><p>“' + b.text + '”</p><cite>— ' + b.cite + '</cite></blockquote>';
      if (b.type === "compare") {
        return '<div class="box compare"><p class="box-label">Quadro comparativo</p><table><thead><tr>' +
          b.headers.map(function (h) { return '<th scope="col">' + h + '</th>'; }).join("") + '</tr></thead><tbody>' +
          b.rows.map(function (r) { return '<tr>' + r.map(function (c) { return '<td>' + c + '</td>'; }).join("") + '</tr>'; }).join("") +
          '</tbody></table></div>';
      }
      return "";
    }

    function renderMain(stagger) {
      var ch = chapter(state.ch);
      applyTheme(ch);
      var idx = CHAPTERS.indexOf(ch);
      var prev = CHAPTERS[idx - 1], next = CHAPTERS[idx + 1];
      var cards = ch.topics.map(function (t, i) {
        var k = key(ch, i), done = !!state.done[k];
        return '<li class="card reveal' + (done ? " is-done" : "") + '" data-key="' + k + '" style="--i:' + (i + 3) + '">' +
          '<button class="card-btn" type="button" data-open="' + i + '" aria-haspopup="dialog">' +
            '<span class="card-media">' + imgTag(t, 640) + '</span>' +
            '<span class="card-body">' +
              '<span class="card-num num">N.º ' + pad(i + 1) + '</span>' +
              '<span class="card-title">' + t.title + '</span>' +
              '<span class="card-excerpt">' + esc(excerpt(t)) + '</span>' +
              '<span class="card-foot"><span class="status">' + (done ? "✓ Concluído" : "Pendente") + '</span>' +
              '<span class="open-hint">Abrir dossiê <svg class="ic" viewBox="0 0 24 24" aria-hidden="true"><path d="M5 12h14M13 6l6 6-6 6"/></svg></span></span>' +
            '</span>' +
            '<span class="card-stamp" aria-hidden="true">✓ Concluído</span>' +
          '</button></li>';
      }).join("");

      var vids = (VIDEOS[ch.id] || []).map(function (v) {
        return '<figure class="video-card" data-video="' + esc(v.id) + '" data-title="' + esc(v.title) + '">' +
          '<div class="video-frame"><button class="video-poster" type="button" data-play aria-label="Assistir: ' + esc(v.title) + '">' +
            '<img src="https://i.ytimg.com/vi/' + esc(v.id) + '/hqdefault.jpg" alt="" loading="lazy" referrerpolicy="no-referrer" onerror="dossieThumbFail(this)">' +
            '<span class="play"><svg viewBox="0 0 24 24" aria-hidden="true"><path d="M7 4.5v15a1 1 0 0 0 1.5.86l12-7.5a1 1 0 0 0 0-1.72l-12-7.5A1 1 0 0 0 7 4.5z"/></svg></span>' +
          '</button></div>' +
          '<figcaption class="video-meta"><span class="video-channel">' + v.channel + '</span><p class="video-title">' + v.title + '</p>' +
            '<p class="video-note">A capa deste vídeo não carregou: a rede pode estar bloqueando o YouTube.</p>' +
            '<a class="btn small" href="https://www.youtube.com/watch?v=' + esc(v.id) + '" target="_blank" rel="noopener noreferrer">Abrir no YouTube ↗</a>' +
          '</figcaption></figure>';
      }).join("");

      $("main").innerHTML =
        '<div class="' + (stagger ? "stagger" : "") + '">' +
          '<header class="chap-head reveal" style="--i:0">' +
            '<span class="chap-kicker">Capítulo ' + ch.num + ' · Dossiê</span>' +
            '<h2>' + ch.title + '</h2>' +
            '<div class="objectives"><p class="box-label">Objetivos de aprendizagem</p><ol>' +
              ch.objectives.map(function (o) { return '<li>' + o + '</li>'; }).join("") + '</ol></div>' +
            '<div class="chap-progress"><span class="line"><span id="chapLine"></span></span><span class="num" id="chapCount"></span></div>' +
          '</header>' +
          '<div class="grid-head reveal" style="--i:1"><h3>Tópicos do dossiê</h3><span class="num" id="gridInfo"></span></div>' +
          '<ul class="grid" id="grid">' + cards + '</ul>' +
          '<p class="empty" id="empty"></p>' +
          '<section class="videos reveal" style="--i:9" aria-label="Videoaulas"><div class="grid-head"><h3>Videoaulas do capítulo</h3><span>Clique para assistir aqui</span></div><div class="video-grid">' + vids + '</div></section>' +
          '<nav class="chapter-nav reveal" style="--i:10">' +
            (prev ? '<button class="btn" type="button" data-goto="' + prev.id + '">← Cap. ' + prev.num + '</button>' : '<span></span>') +
            (next ? '<button class="btn primary" type="button" data-goto="' + next.id + '">Cap. ' + next.num + ' →</button>' : '<span></span>') +
          '</nav>' +
        '</div>';
      applyFilter();
      updateProgress(false);
    }

    window.dossieThumbFail = function (img) {
      var card = img.closest(".video-card"), poster = img.closest(".video-poster");
      if (!card || !poster) return;
      card.classList.add("thumb-missing"); poster.classList.add("no-thumb");
      var t = document.createElement("span"); t.className = "poster-title"; t.textContent = card.getAttribute("data-title");
      poster.replaceChild(t, img);
    };

    function applyFilter() {
      var cards = document.querySelectorAll("#grid .card"), shown = 0;
      for (var i = 0; i < cards.length; i++) {
        var done = cards[i].classList.contains("is-done");
        var vis = state.filter === "all" || (state.filter === "done" ? done : !done);
        cards[i].hidden = !vis; if (vis) shown++;
      }
      var empty = $("empty");
      if (empty) {
        empty.classList.toggle("show", shown === 0);
        empty.textContent = state.filter === "todo" ? "Nenhum tópico pendente neste capítulo — dossiê completo!" : "Você ainda não carimbou nenhum tópico deste capítulo.";
        var labels = { all: "Todos os tópicos", todo: "Somente pendentes", done: "Somente concluídos" };
        $("gridInfo").textContent = labels[state.filter] + " · " + shown + " de " + cards.length;
      }
    }

    function selectChapter(id, opts) {
      if (!chapter(id) || id === state.ch) return;
      state.ch = id; save();
      var b = document.querySelectorAll(".ch-btn");
      for (var i = 0; i < b.length; i++) b[i].setAttribute("aria-current", b[i].getAttribute("data-ch") === id);
      renderMain(!(opts && opts.keyboard));
      var top = $("main").getBoundingClientRect().top + window.pageYOffset;
      if (window.pageYOffset > top) window.scrollTo({ top: top, behavior: reduceMotion ? "auto" : "smooth" });
    }

    /* =====================================================================
       GAVETA
       ===================================================================== */
    var lastFocus = null;
    function fillDrawer(i) {
      var ch = chapter(state.ch), t = ch.topics[i], k = key(ch, i), done = !!state.done[k];
      drawerIndex = i;
      $("drawerTag").textContent = "Cap. " + ch.num + " · N.º " + pad(i + 1) + " de " + pad(ch.topics.length);
      $("drawerBody").innerHTML =
        '<figure class="figure"><button class="figure-btn" type="button" data-zoom aria-label="Ampliar obra">' + imgTag(t, 1200) + '</button><figcaption>' + t.caption + '</figcaption></figure>' +
        '<h2 class="drawer-title" id="drawerTitle">' + t.title + '</h2>' +
        '<div class="prose">' + t.text.map(function (p) { return '<p>' + p + '</p>'; }).join("") + '</div>' +
        (t.boxes || []).map(renderBox).join("") +
        '<div class="stamp-wrap"><button class="stamp" type="button" data-stamp="' + k + '" aria-pressed="' + done + '">' + (done ? "[ ✓ Concluído ]" : "[ Carimbar como estudado ]") + '</button>' +
        '<span class="stamp-hint">' + (done ? "Clique de novo para desfazer." : "Marque quando terminar a leitura.") + '</span></div>';
      $("drawerPrev").disabled = i === 0;
      $("drawerNext").disabled = i === ch.topics.length - 1;
      $("drawerScroll").scrollTop = 0;
    }
    function openDrawer(i) {
      lastFocus = document.activeElement;
      fillDrawer(i);
      var d = $("drawer");
      d.classList.add("is-open"); d.setAttribute("aria-hidden", "false");
      document.body.classList.add("locked");
      setTimeout(function () { var c = d.querySelector("[data-close-drawer].btn"); if (c) c.focus({ preventScroll: true }); }, 0);
    }
    function closeDrawer() {
      var d = $("drawer");
      if (!d.classList.contains("is-open")) return;
      d.classList.remove("is-open"); d.setAttribute("aria-hidden", "true");
      document.body.classList.remove("locked");
      if (lastFocus && lastFocus.focus) lastFocus.focus({ preventScroll: true });
    }
    function stepDrawer(dir) {
      var ch = chapter(state.ch), n = drawerIndex + dir;
      if (n < 0 || n >= ch.topics.length) return;
      var body = $("drawerBody");
      if (reduceMotion) { fillDrawer(n); return; }
      body.classList.add("swap");
      setTimeout(function () { fillDrawer(n); body.classList.remove("swap"); }, 160);
    }

    function toggleDone(k) {
      var on = !state.done[k];
      if (on) state.done[k] = true; else delete state.done[k];
      save();
      var stamp = document.querySelector('[data-stamp="' + k + '"]');
      if (stamp) {
        stamp.setAttribute("aria-pressed", on);
        stamp.textContent = on ? "[ ✓ Concluído ]" : "[ Carimbar como estudado ]";
        stamp.nextElementSibling.textContent = on ? "Clique de novo para desfazer." : "Marque quando terminar a leitura.";
        stamp.classList.remove("pressing");
        if (on) { void stamp.offsetWidth; stamp.classList.add("pressing"); }
      }
      var card = document.querySelector('.card[data-key="' + k + '"]');
      if (card) {
        card.classList.toggle("is-done", on);
        card.querySelector(".status").textContent = on ? "✓ Concluído" : "Pendente";
      }
      applyFilter();
      updateProgress(true);
      var ch = chapter(state.ch);
      if (on && doneCount(ch) === ch.topics.length) toast("Capítulo " + ch.num + " concluído — dossiê completo!");
    }

    /* =====================================================================
       MODAIS
       ===================================================================== */
    var modalFocus = null;
    function openModal(id) {
      modalFocus = document.activeElement;
      var m = $(id); m.classList.add("is-open"); m.setAttribute("aria-hidden", "false");
      setTimeout(function () { var b = m.querySelector("button"); if (b) b.focus({ preventScroll: true }); }, 0);
    }
    function closeModal(id) {
      var m = $(id); if (!m.classList.contains("is-open")) return false;
      m.classList.remove("is-open"); m.setAttribute("aria-hidden", "true");
      if (modalFocus && modalFocus.focus) modalFocus.focus({ preventScroll: true });
      return true;
    }

    function playVideo(card) {
      if (!card || card.querySelector("iframe")) return;
      var id = card.getAttribute("data-video"), frame = card.querySelector(".video-frame"), poster = card.querySelector(".video-poster");
      var f = document.createElement("iframe");
      f.className = "video-iframe";
      f.src = "https://www.youtube-nocookie.com/embed/" + encodeURIComponent(id) + "?autoplay=1&rel=0";
      f.title = card.getAttribute("data-title");
      f.setAttribute("allow", "accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture");
      f.setAttribute("allowfullscreen", "");
      f.setAttribute("referrerpolicy", "strict-origin-when-cross-origin"); /* o YouTube exige o referenciador */
      frame.appendChild(f);
      requestAnimationFrame(function () { f.classList.add("is-ready"); if (poster) poster.classList.add("is-leaving"); });
      setTimeout(function () { if (poster && poster.parentNode) poster.parentNode.removeChild(poster); }, 300);
    }

    /* =====================================================================
       EVENTOS
       ===================================================================== */
    function bind() {
      $("chapters").addEventListener("click", function (e) {
        var b = e.target.closest(".ch-btn"); if (b) selectChapter(b.getAttribute("data-ch"), { keyboard: e.detail === 0 });
      });
      $("filters").addEventListener("click", function (e) {
        var b = e.target.closest("[data-filter]"); if (!b) return;
        state.filter = b.getAttribute("data-filter"); save();
        var all = $("filters").querySelectorAll("button");
        for (var i = 0; i < all.length; i++) all[i].setAttribute("aria-pressed", all[i] === b);
        applyFilter();
      });
      $("main").addEventListener("click", function (e) {
        var o = e.target.closest("[data-open]"); if (o) { openDrawer(parseInt(o.getAttribute("data-open"), 10)); return; }
        var p = e.target.closest("[data-play]"); if (p) { playVideo(p.closest(".video-card")); return; }
        var g = e.target.closest("[data-goto]"); if (g) selectChapter(g.getAttribute("data-goto"));
      });
      $("drawer").addEventListener("click", function (e) {
        if (e.target.closest("[data-close-drawer]")) { closeDrawer(); return; }
        var s = e.target.closest("[data-stamp]"); if (s) { toggleDone(s.getAttribute("data-stamp")); return; }
        var z = e.target.closest("[data-zoom]");
        if (z) {
          var img = z.querySelector("img"), cap = z.parentNode.querySelector("figcaption");
          if (img) { $("lbImg").src = img.currentSrc || img.src; $("lbImg").alt = img.alt; $("lbCap").textContent = cap ? cap.textContent : img.alt; openModal("lightbox"); }
        }
      });
      $("drawerPrev").addEventListener("click", function () { stepDrawer(-1); });
      $("drawerNext").addEventListener("click", function () { stepDrawer(1); });
      $("btnReset").addEventListener("click", function () { openModal("resetDialog"); });
      $("resetConfirm").addEventListener("click", function () {
        state.done = {}; save(); closeModal("resetDialog"); renderMain(false); toast("Progresso reiniciado");
      });
      document.addEventListener("click", function (e) {
        var c = e.target.closest("[data-close-modal]"); if (c) closeModal(c.closest(".modal").id);
      });
      $("lbImg").addEventListener("click", function () { closeModal("lightbox"); });
      document.addEventListener("keydown", function (e) {
        if (e.key === "Escape") {
          if (closeModal("lightbox") || closeModal("resetDialog")) return;
          closeDrawer();
        }
        if ($("drawer").classList.contains("is-open") && !$("lightbox").classList.contains("is-open")) {
          if (e.key === "ArrowRight") fillDrawerSafe(1);
          if (e.key === "ArrowLeft") fillDrawerSafe(-1);
        }
      });
    }
    /* Teclado troca de tópico sem animação (ação repetida) */
    function fillDrawerSafe(dir) {
      var ch = chapter(state.ch), n = drawerIndex + dir;
      if (n >= 0 && n < ch.topics.length) fillDrawer(n);
    }

    /* =====================================================================
       INÍCIO
       ===================================================================== */
    function init() {
      load();
      if (!chapter(state.ch)) state.ch = "c11";
      renderSidebar();
      renderMain(true);
      bind();
    }
    if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", init); else init();
  })();
  </script>
