    function loadState() {
      try {
        window.localStorage.setItem("__t__", "1");
        window.localStorage.removeItem("__t__");
      } catch (e) {
        storageOK = false;
        $("storageWarn").classList.add("show");
        return;
      }
      try {
        var raw = window.localStorage.getItem(STORAGE_KEY);
        if (!raw) {
          /* Versão anterior tinha outros tópicos: aproveita só as anotações */
          var old = window.localStorage.getItem(OLD_KEY);
          if (old) { var o = JSON.parse(old); if (o && o.notes && typeof o.notes === "object") { state.notes = o.notes; window.localStorage.setItem(STORAGE_KEY, JSON.stringify(state)); } }
          return;
        }
        var s = JSON.parse(raw);
        if (s && typeof s === "object") {
          state.read = s.read && typeof s.read === "object" ? s.read : {};
          state.notes = s.notes && typeof s.notes === "object" ? s.notes : {};
          state.tab = typeof s.tab === "string" ? s.tab : "c11";
        }
      } catch (e) { /* dados inválidos: recomeça */ }
    }

    /* Contagem animada (count-up) */
    function countTo(el, to) {
      if (!el) return;
      var from = parseInt(el.getAttribute("data-val") || el.textContent, 10) || 0;
      el.setAttribute("data-val", to);
      if (reduceMotion || from === to) { el.textContent = to; return; }
      var start = null, dur = 420;
      var token = {};
      el._countToken = token;
      function frame(ts) {
        if (el._countToken !== token) return;
        if (start === null) start = ts;
        var p = Math.min(1, (ts - start) / dur);
        var eased = 1 - Math.pow(1 - p, 3);
        el.textContent = Math.round(from + (to - from) * eased);
        if (p < 1) window.requestAnimationFrame(frame);
      }
      window.requestAnimationFrame(frame);
    }

    function pulse(el) {
      if (!el || reduceMotion) return;
      el.classList.remove("pulse");
      void el.offsetWidth;
      el.classList.add("pulse");
    }

    function updateProgress(animate) {
      var total = 0, done = 0;
      CHAPTERS.forEach(function (ch) {
        var p = progressOf(ch);
        total += p.total; done += p.done;
        var c = document.querySelector('.tab[data-ch="' + ch.id + '"] .tab-count');
        if (c) c.textContent = p.done + "/" + p.total;
      });
      var pct = total ? Math.round((done / total) * 100) : 0;
      if (animate) countTo($("overallPct"), pct); else { $("overallPct").textContent = pct; $("overallPct").setAttribute("data-val", pct); }
      $("overallMeta").textContent = done + " de " + total + " tópicos";
      $("overallBar").setAttribute("aria-valuenow", pct);
      $("overallBar").querySelector(".fill").style.transform = "scaleX(" + (total ? done / total : 0) + ")";

      var ch = chapterById(state.tab);
      var cp = progressOf(ch);
      var d = $("chapDone");
      if (d) {
        if (animate) countTo(d, cp.done); else { d.textContent = cp.done; d.setAttribute("data-val", cp.done); }
        var cpPct = Math.round(cp.ratio * 100);
        if (animate) countTo($("chapPct"), cpPct); else { $("chapPct").textContent = cpPct; $("chapPct").setAttribute("data-val", cpPct); }
        $("chapProgBar").setAttribute("aria-valuenow", cpPct);
        $("chapProgBar").querySelector(".fill").style.transform = "scaleX(" + cp.ratio + ")";
      }
    }

    function renderBox(b) {
      if (b.type === "quote") {
        return '<blockquote class="quote"><p>“' + b.text + '”</p><cite>— ' + b.cite + '</cite></blockquote>';
      }
      if (b.type === "concept") {
        return '<div class="callout concepts"><p class="callout-label">Conceitos-chave</p><dl>' +
          b.items.map(function (it) { return '<div><dt>' + it[0] + '</dt><dd>' + it[1] + '</dd></div>'; }).join("") +
          '</dl></div>';
      }
      if (b.type === "compare") {
        var head = b.headers.map(function (h) { return '<th scope="col">' + h + '</th>'; }).join("");
        var rows = b.rows.map(function (r) {
          return '<tr>' + r.map(function (c, i) { return '<td data-label="' + esc(b.headers[i]) + '">' + c + '</td>'; }).join("") + '</tr>';
        }).join("");
        return '<div class="callout compare"><p class="callout-label">Quadro comparativo</p>' +
          (b.title ? '<p class="compare-title">' + b.title + '</p>' : '') +
          '<table><thead><tr>' + head + '</tr></thead><tbody>' + rows + '</tbody></table></div>';
      }
      if (b.type === "timeline") {
        return '<div class="callout timeline"><p class="callout-label">Linha do tempo</p><ol>' +
          b.items.map(function (it) { return '<li><span class="when">' + it[0] + '</span><span>' + it[1] + '</span></li>'; }).join("") +
          '</ol></div>';
      }
      return "";
    }

    /* 1ª falha: tenta de novo pelo nome do arquivo (Special:FilePath resolve o
       caminho correto no acervo). 2ª falha: placa museológica. */
    window.imgFallback = function (img) {
      try {
        var btn = img.closest(".figure-btn");
        if (!btn) return;
        var original = img.getAttribute("data-original") || img.getAttribute("src") || "";
        if (!img.getAttribute("data-retried") && original.indexOf("upload.wikimedia.org") > -1) {
          img.setAttribute("data-retried", "1");
          img.setAttribute("data-original", original);
          img.src = "https://commons.wikimedia.org/wiki/Special:FilePath/" + commonsFile(original) + "?width=1200";
          return;
        }
        var cap = img.getAttribute("data-caption") || "Obra histórica";
        var src = original;
        var fig = btn.parentNode;
        var plate = document.createElement("div");
        plate.className = "placard";
        plate.innerHTML =
          '<p class="placard-title">' + esc(cap) + '</p>' +
          (src ? '<a class="placard-link" href="' + esc(commonsPage(src)) + '" target="_blank" rel="noopener noreferrer">Ver acervo original ↗</a>' : '');
        fig.replaceChild(plate, btn);
        var fc = fig.querySelector("figcaption");
        if (fc) fc.style.display = "none";
      } catch (e) { /* silencioso */ }
    };

            '<button class="quiet-btn" type="button" data-all="close">' + I.collapse + 'Recolher <span class="lbl-long">todos</span></button>' +
            '<button class="quiet-btn" type="button" id="notesToggle" aria-expanded="' + notesOpen + '" aria-controls="notesDrawer">' + I.note + 'Anotações' + I.chev + '</button>' +
          '</div>' +

          '<div class="collapse notes-drawer' + (notesOpen ? " is-open" : "") + '" id="notesDrawer">' +
            '<div class="collapse-inner"><div class="notes-card">' +
              '<header><h4>Minhas anotações — Cap. ' + ch.num + '</h4><p class="notes-hint">Opcional · salvas automaticamente</p></header>' +
              '<label class="sr-only" for="notesArea">Anotações do capítulo ' + ch.num + '</label>' +
              '<textarea id="notesArea" placeholder="Resumos, dúvidas, datas e conexões entre os tópicos deste capítulo…">' + esc(state.notes[ch.id] || "") + '</textarea>' +
              '<div class="notes-foot">' +
                '<span class="save-state" id="saveState"><span class="dot"></span><span id="saveTxt">Pronto</span> · <span class="num" id="charCount">0</span> caracteres</span>' +
                '<span class="notes-actions">' +
                  '<button class="quiet-btn" type="button" id="copyNotes">' + I.copy + 'Copiar</button>' +
                  '<button class="quiet-btn" type="button" id="copyAllNotes">' + I.copy + 'Copiar todas</button>' +
                '</span>' +
              '</div>' +
            '</div></div>' +
          '</div>' +

          '<ol class="toc">' + ch.topics.map(function (t, i) { return renderTopic(ch, t, i); }).join("") + '</ol>' +

          renderVideos(ch) +

          '<nav class="chapter-nav reveal" style="--i:14" aria-label="Navegação entre capítulos">' +
            (prev ? '<button class="nav-link" type="button" data-goto="' + prev.id + '"><small>← Anterior</small><span>Cap. ' + prev.num + ' · ' + prev.short + '</span></button>' : '') +
            (next ? '<button class="nav-link next" type="button" data-goto="' + next.id + '"><small>Próximo →</small><span>Cap. ' + next.num + ' · ' + next.short + '</span></button>' : '') +
          '</nav>' +
        '</div>';

      $("chapterView").innerHTML = html;
      updateCharCount();
      updateProgress(false);
    }

    function selectTab(id, opts) {
      opts = opts || {};
      if (!chapterById(id) || id === state.tab) return;
      state.tab = id;
      saveState();
      var tabs = document.querySelectorAll(".tab");
      for (var i = 0; i < tabs.length; i++) {
        var on = tabs[i].getAttribute("data-ch") === id;
        tabs[i].setAttribute("aria-selected", on);
        tabs[i].setAttribute("tabindex", on ? "0" : "-1");
      }
      /* Ações de teclado não animam: a troca é instantânea */
      positionIndicator(!opts.keyboard);
      renderChapter(!opts.keyboard);
      if (opts.scroll) {
        var y = $("tabs").getBoundingClientRect().top + window.pageYOffset - 8;
        if (window.pageYOffset > y) {
          try { window.scrollTo({ top: y, behavior: reduceMotion ? "auto" : "smooth" }); } catch (e) { window.scrollTo(0, y); }
        }
      }
    }

    /* =====================================================================
       ACCORDION
       ===================================================================== */
    function setTopic(li, open) {
      openTopics[li.getAttribute("data-key")] = open;
      li.classList.toggle("is-open", open);
      li.querySelector(".topic-trigger").setAttribute("aria-expanded", open);
    }
    function setAll(open) {
      var list = document.querySelectorAll("#chapterView .topic");
      for (var i = 0; i < list.length; i++) setTopic(list[i], open);
    }

    function toggleRead(key) {
      var on = !state.read[key];
      if (on) state.read[key] = true; else delete state.read[key];
      saveState();
      var li = document.querySelector('.topic[data-key="' + key + '"]');
      if (li) {
        li.classList.toggle("is-read", on);
        li.querySelector(".topic-status").textContent = on ? "Lido" : "A ler";
        var btns = li.querySelectorAll("[data-read]");
        for (var i = 0; i < btns.length; i++) btns[i].setAttribute("aria-pressed", on);
        var lbl = li.querySelector(".read-label");
        if (lbl) lbl.textContent = on ? "Lido" : "Marcar como lido";
      }
      updateProgress(true);
      if (on) {
        pulse($("chapProgBar"));
        var ch = chapterById(state.tab), p = progressOf(ch);
        if (p.done === p.total) toast("Capítulo " + ch.num + " concluído — excelente leitura!");
      }
    }

    /* =====================================================================
       ANOTAÇÕES
       ===================================================================== */
    var saveTimer = null;
    function updateCharCount() {
      var ta = $("notesArea");
      if (ta) $("charCount").textContent = ta.value.length;
    }
    function copyText(text) {
      return new Promise(function (resolve) {
        function legacy() {
          var ta = document.createElement("textarea");
          ta.value = text;
          ta.setAttribute("readonly", "");
          ta.style.position = "fixed"; ta.style.top = "0"; ta.style.opacity = "0";
          document.body.appendChild(ta);
          ta.select();
          var ok = false;
          try { ok = document.execCommand("copy"); } catch (e) {}
          document.body.removeChild(ta);
          resolve(ok);
        }
        try {
          if (navigator.clipboard && window.isSecureContext) navigator.clipboard.writeText(text).then(function () { resolve(true); }, legacy);
          else legacy();
        } catch (e) { legacy(); }
      });
    }

    /* =====================================================================
       MODAIS
       ===================================================================== */
    var lastFocus = null;
    function openModal(id) {
      lastFocus = document.activeElement;
      var m = $(id);
      m.classList.add("is-open");
      m.setAttribute("aria-hidden", "false");
      var f = m.querySelector(".modal-close, button");
      if (f) setTimeout(function () { f.focus({ preventScroll: true }); }, 0);
    }
    function closeModal(id) {
      var m = $(id);
      if (!m.classList.contains("is-open")) return;
      m.classList.remove("is-open");
      m.setAttribute("aria-hidden", "true");
      if (lastFocus && lastFocus.focus) lastFocus.focus({ preventScroll: true });
    }
    function openLightbox(src, cap) {
      var img = $("lbImg");
      img.setAttribute("referrerpolicy", "no-referrer");
      img.src = src;
      img.alt = cap || "";
      $("lbCap").textContent = cap || "";
      openModal("lightbox");
    }

    /* =====================================================================
       EVENTOS
       ===================================================================== */
    function bind() {
      var tabsEl = $("tabs");
      tabsEl.addEventListener("click", function (e) {
        var b = e.target.closest(".tab");
        if (b) selectTab(b.getAttribute("data-ch"), { keyboard: e.detail === 0 });
      });
      tabsEl.addEventListener("keydown", function (e) {
        var k = e.key;
        if (k !== "ArrowRight" && k !== "ArrowLeft" && k !== "Home" && k !== "End") return;
        e.preventDefault();
        var idx = CHAPTERS.indexOf(chapterById(state.tab));
        if (k === "ArrowRight") idx = (idx + 1) % CHAPTERS.length;
        if (k === "ArrowLeft") idx = (idx - 1 + CHAPTERS.length) % CHAPTERS.length;
        if (k === "Home") idx = 0;
        if (k === "End") idx = CHAPTERS.length - 1;
        selectTab(CHAPTERS[idx].id, { keyboard: true });
        $("tab-" + CHAPTERS[idx].id).focus();
      });

      var view = $("chapterView");
      view.addEventListener("click", function (e) {
        var trig = e.target.closest(".topic-trigger");
        if (trig) { var li = trig.closest(".topic"); setTopic(li, !li.classList.contains("is-open")); return; }

        var play = e.target.closest("[data-play]");
        if (play) { playVideo(play.closest(".video-card")); return; }

        var zoom = e.target.closest("[data-zoom]");
        if (zoom) { var img = zoom.querySelector("img"); if (img) openLightbox(img.currentSrc || img.getAttribute("src"), img.getAttribute("data-caption")); return; }

        var rd = e.target.closest("[data-read]");
        if (rd) { toggleRead(rd.getAttribute("data-read")); return; }

        var all = e.target.closest("[data-all]");
        if (all) { setAll(all.getAttribute("data-all") === "open"); return; }

        if (e.target.closest("#notesToggle")) {
          notesOpen = !notesOpen;
          $("notesDrawer").classList.toggle("is-open", notesOpen);
          $("notesToggle").setAttribute("aria-expanded", notesOpen);
          if (notesOpen) setTimeout(function () { var ta = $("notesArea"); if (ta) ta.focus({ preventScroll: true }); }, 0);
          return;
        }

        if (e.target.closest("#copyNotes")) {
          var ch = chapterById(state.tab), txt = (state.notes[ch.id] || "").trim();
          if (!txt) { toast("Ainda não há anotações neste capítulo."); return; }
          copyText("Capítulo " + ch.num + " — " + ch.title + "\n\n" + txt).then(function (ok) {
            if (ok) toast("Anotações do Cap. " + ch.num + " copiadas");
            else { $("notesArea").select(); toast("Texto selecionado: use Ctrl+C para copiar"); }
          });
          return;
        }
        if (e.target.closest("#copyAllNotes")) {
          var parts = [];
          CHAPTERS.forEach(function (c) {
            var t = (state.notes[c.id] || "").trim();
            if (t) parts.push("— Capítulo " + c.num + ": " + c.title + " —\n" + t);
          });
          if (!parts.length) { toast("Você ainda não fez anotações."); return; }
          copyText("Plano de Estudos de História — Minhas anotações\n\n" + parts.join("\n\n")).then(function (ok) {
            toast(ok ? "Todas as anotações foram copiadas" : "Não foi possível copiar neste navegador");
          });
          return;
        }

        var go = e.target.closest("[data-goto]");
        if (go) selectTab(go.getAttribute("data-goto"), { scroll: true });
      });

      view.addEventListener("input", function (e) {
        if (e.target.id !== "notesArea") return;
        var ch = chapterById(state.tab);
        state.notes[ch.id] = e.target.value;
        updateCharCount();
        $("saveState").classList.remove("saved");
        $("saveTxt").textContent = "Salvando…";
        clearTimeout(saveTimer);
        saveTimer = setTimeout(function () {
          var ok = saveState();
          var ss = $("saveState");
          if (!ss) return;
          ss.classList.toggle("saved", ok);
          $("saveTxt").textContent = ok ? "Salvo" : "Salvo só nesta sessão";
        }, 400);
      });

      $("btnReset").addEventListener("click", function () { $("resetNotes").checked = false; openModal("resetDialog"); });
      $("resetConfirm").addEventListener("click", function () {
        state.read = {};
        if ($("resetNotes").checked) state.notes = {};
        openTopics = {};
        saveState();
        closeModal("resetDialog");
        renderChapter(true);
        toast("Leitura reiniciada");
      });

      document.addEventListener("click", function (e) {
        var c = e.target.closest("[data-close]");
        if (!c) return;
        var m = c.closest(".modal");
        if (m) closeModal(m.id);
      });
      $("lbImg").addEventListener("click", function () { closeModal("lightbox"); });
      $("lbImg").addEventListener("error", function () { if ($("lightbox").classList.contains("is-open")) closeModal("lightbox"); });
      document.addEventListener("keydown", function (e) {
        if (e.key === "Escape") { closeModal("lightbox"); closeModal("resetDialog"); }
      });

      var rt = null;
      window.addEventListener("resize", function () {
        clearTimeout(rt);
        rt = setTimeout(function () { positionIndicator(false); }, 80);
      });
    }

    /* =====================================================================
       INICIALIZAÇÃO
       ===================================================================== */
    function init() {
      loadState();
      if (!chapterById(state.tab)) state.tab = "c11";
      renderTabs();
      renderChapter(true);
      bind();
      positionIndicator(false);
      if (document.fonts && document.fonts.ready) document.fonts.ready.then(function () { positionIndicator(false); });
    }

    if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", init);
    else init();
  })();
